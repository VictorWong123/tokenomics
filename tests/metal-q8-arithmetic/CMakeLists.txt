cmake_minimum_required(VERSION 3.24)
project(tokenomics_q8_arithmetic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${root}/third_party/llama.cpp"
  OUTPUT_VARIABLE pin OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT pin STREQUAL "fd27b2034905f8dadd8bd6d2cc44595f4929d391")
  message(FATAL_ERROR "Review Q8 reference extraction for the changed upstream pin")
endif()
execute_process(COMMAND git status --porcelain --untracked-files=normal
  WORKING_DIRECTORY "${root}/third_party/llama.cpp"
  OUTPUT_VARIABLE dirty OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT dirty STREQUAL "")
  message(FATAL_ERROR "The source-derived Q8 reference requires a clean upstream submodule")
endif()
include("${root}/cmake/source-input.cmake")
tokenomics_read_source("${root}/third_party/llama.cpp/ggml/src/ggml-metal/kernels/mul_mv.metal" source)
string(FIND "${source}" "template<short NR0, typename args_t>\nvoid kernel_mul_mv_q8_0_f32_impl(" start)
string(FIND "${source}" "[[host_name(\"kernel_mul_mv_q8_0_f32\")]]" finish)
if(start LESS 0 OR finish LESS start)
  message(FATAL_ERROR "Pinned Q8 function boundaries changed")
endif()
math(EXPR length "${finish} - ${start}")
string(SUBSTRING "${source}" ${start} ${length} reference)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/q8-reference.inc" "${reference}")
option(TOKENOMICS_ARITHMETIC_SANITIZE "Run source-derived lane arithmetic with sanitizers" OFF)
foreach(mode strict contracted)
  add_executable(test_q8_${mode} "${root}/tests/test_metal_q8_arithmetic.cpp")
  target_include_directories(test_q8_${mode} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
    "${root}/third_party/llama.cpp/ggml/src/ggml-metal")
  if(mode STREQUAL "strict")
    target_compile_options(test_q8_${mode} PRIVATE -fno-fast-math -ffp-contract=off)
  else()
    target_compile_options(test_q8_${mode} PRIVATE -ffp-contract=fast)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
      target_compile_options(test_q8_${mode} PRIVATE -mcpu=native)
    else()
      target_compile_options(test_q8_${mode} PRIVATE -march=native)
    endif()
  endif()
  if(TOKENOMICS_ARITHMETIC_SANITIZE)
    target_compile_options(test_q8_${mode} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(test_q8_${mode} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
enable_testing()
add_test(NAME q8_strict_partials COMMAND test_q8_strict)
add_test(NAME q8_contracted_partials COMMAND test_q8_contracted)
