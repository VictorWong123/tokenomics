cmake_minimum_required(VERSION 3.24)
project(tokenomics_metal_host_check LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
# Linux-only host integration check, not a Metal backend or inference build.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Use the main project's native Metal tests on Apple platforms")
endif()
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(TOKENOMICS_CPU_BUILD "${root}/build-cpu" CACHE PATH "Existing Linux CPU build providing ggml-base")
set(base "${TOKENOMICS_CPU_BUILD}/third_party/llama.cpp/ggml/src/libggml-base.a")
if(NOT EXISTS "${base}")
  message(FATAL_ERROR "Build the default CPU profile before checking Metal host code")
endif()
set(metal "${root}/third_party/llama.cpp/ggml/src/ggml-metal")
add_library(ggml-metal STATIC "${metal}/ggml-metal-common.cpp" "${metal}/ggml-metal-ops.cpp")
include("${root}/cmake/metal-short-mv.cmake")
tokenomics_metal_short_mv("${root}")
option(TOKENOMICS_HOST_INTERLEAVE "Compile interleaved host dispatch without a GPU" OFF)
if(TOKENOMICS_HOST_INTERLEAVE)
  set(TOKENOMICS_METAL_SHORT_MV ON)
  set(GGML_METAL_EMBED_LIBRARY ON)
  set(original_asm "${CMAKE_CURRENT_BINARY_DIR}/ggml-metal-embed-mul_mv.s")
  file(WRITE "${original_asm}" "")
  target_sources(ggml-metal PRIVATE "${metal}/ggml-metal-device.cpp" "${original_asm}")
  include("${root}/cmake/metal-interleave.cmake")
  tokenomics_metal_interleave("${root}")
  # Mach-O assembly is retained for inspection; Linux only compiles C++ here.
  set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/tokenomics-metal-embed-mul_mv.s"
    PROPERTIES HEADER_FILE_ONLY TRUE)
endif()
target_include_directories(ggml-metal PRIVATE "${root}/third_party/llama.cpp/ggml/include" "${root}/third_party/llama.cpp/ggml/src")
target_compile_options(ggml-metal PRIVATE -ffunction-sections -fdata-sections)
find_package(OpenMP REQUIRED)
find_package(Threads REQUIRED)
add_executable(test_metal_dispatch "${root}/tests/test_metal_dispatch.cpp")
target_include_directories(test_metal_dispatch PRIVATE "${root}/third_party/llama.cpp/ggml/include" "${metal}")
target_link_libraries(test_metal_dispatch PRIVATE ggml-metal "${base}" OpenMP::OpenMP_CXX Threads::Threads)
# Unused host functions refer to the unavailable GPU runtime. Keep only the
# dispatch predicates exercised here; both complete translation units compile.
target_link_options(test_metal_dispatch PRIVATE -Wl,--gc-sections)
enable_testing()
add_test(NAME metal_host_dispatch COMMAND test_metal_dispatch)

include("${root}/cmake/metal-context.cmake")
foreach(mode single-row short-mv)
  set(generated "${CMAKE_CURRENT_BINARY_DIR}/tokenomics-context-${mode}.cpp")
  tokenomics_write_metal_context("${root}" "${generated}" "${mode}")
  add_library(context-${mode}-compile OBJECT "${generated}")
  target_include_directories(context-${mode}-compile PRIVATE
    "${root}/third_party/llama.cpp/src"
    "${root}/third_party/llama.cpp/include"
    "${root}/third_party/llama.cpp/ggml/include"
    "${root}/third_party/llama.cpp/ggml/src")
endforeach()
add_executable(test_metal_verification "${root}/tests/test_metal_verification.cpp")
target_include_directories(test_metal_verification PRIVATE "${root}/src")
add_test(NAME metal_verification_split COMMAND test_metal_verification)
