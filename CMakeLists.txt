cmake_minimum_required(VERSION 3.24)
project(tokenomics VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
option(TOKENOMICS_SANITIZE "Enable address/undefined sanitizers (CPU)" OFF)
option(TOKENOMICS_CPU_BATCH_INVARIANT "Use consistent CPU dot kernels across verification batch widths" ON)
option(TOKENOMICS_CPU_PACKED_VERIFY "Experimental row-consistent packed CPU verification" OFF)
option(TOKENOMICS_BUILD_STOCK_DIAGNOSTIC "Build the pinned upstream speculative example for diagnostics" OFF)
if(TOKENOMICS_CPU_BATCH_INVARIANT AND TOKENOMICS_CPU_PACKED_VERIFY)
  message(FATAL_ERROR "Select one CPU profile: disable TOKENOMICS_CPU_BATCH_INVARIANT for TOKENOMICS_CPU_PACKED_VERIFY")
endif()
if(TOKENOMICS_CPU_PACKED_VERIFY)
  set(GGML_CPU_REPACK ON CACHE BOOL "" FORCE)
  set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
  set(GGML_BLAS OFF CACHE BOOL "" FORCE)
elseif(TOKENOMICS_CPU_BATCH_INVARIANT)
  # Greedy speculative verification must use the same arithmetic as a one-row
  # target decode. The pinned backend's repack/SGEMM paths vary with batch width.
  set(GGML_CPU_REPACK OFF CACHE BOOL "" FORCE)
  set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
  set(GGML_BLAS OFF CACHE BOOL "" FORCE)
endif()
if(TOKENOMICS_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/third_party/llama.cpp OUTPUT_VARIABLE LLAMA_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT LLAMA_REV STREQUAL "fd27b2034905f8dadd8bd6d2cc44595f4929d391")
  message(FATAL_ERROR "Wrong llama.cpp revision. Run git submodule update --init --recursive")
endif()
execute_process(COMMAND git status --porcelain --untracked-files=normal
  WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/third_party/llama.cpp
  OUTPUT_VARIABLE LLAMA_STATUS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT LLAMA_STATUS STREQUAL "")
  message(FATAL_ERROR "llama.cpp must be clean: the pinned commit does not identify local source edits")
endif()
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/llama.cpp EXCLUDE_FROM_ALL)
if(TOKENOMICS_CPU_BATCH_INVARIANT OR TOKENOMICS_CPU_PACKED_VERIFY)
  # Keep fast common prefills, but never use tiled arithmetic for the at-most
  # nine target verification rows (pending token plus eight proposals).
  # Build a guarded copy: the pinned submodule itself stays untouched.
  set(TILED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/llama.cpp/ggml/src/ggml-cpu/tiled")
  file(READ "${TILED_DIR}/tiled.cpp" TILED_SOURCE)
  set(TILED_SELECTOR "return rows >= 8 || ggml_tiled_matmul_forced();")
  string(FIND "${TILED_SOURCE}" "${TILED_SELECTOR}" TILED_SELECTOR_POSITION)
  if(TILED_SELECTOR_POSITION EQUAL -1)
    message(FATAL_ERROR "Pinned CPU tiled selector changed; review the verification guard")
  endif()
  string(REPLACE "${TILED_SELECTOR}" "return rows > 9 && (rows >= 8 || ggml_tiled_matmul_forced());" TILED_SOURCE "${TILED_SOURCE}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/tokenomics-tiled.cpp" "${TILED_SOURCE}")
  get_target_property(CPU_SOURCES ggml-cpu SOURCES)
  list(LENGTH CPU_SOURCES CPU_SOURCE_COUNT)
  list(FILTER CPU_SOURCES EXCLUDE REGEX "(^|/)tiled/tiled\\.cpp$")
  list(LENGTH CPU_SOURCES GUARDED_CPU_SOURCE_COUNT)
  math(EXPR EXPECTED_CPU_SOURCE_COUNT "${CPU_SOURCE_COUNT} - 1")
  if(NOT GUARDED_CPU_SOURCE_COUNT EQUAL EXPECTED_CPU_SOURCE_COUNT)
    message(FATAL_ERROR "Could not isolate the pinned CPU tiled source")
  endif()
  set_property(TARGET ggml-cpu PROPERTY SOURCES ${CPU_SOURCES})
  target_sources(ggml-cpu PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/tokenomics-tiled.cpp")
  target_include_directories(ggml-cpu PRIVATE "${TILED_DIR}")
endif()
if(TOKENOMICS_CPU_PACKED_VERIFY)
  include(cmake/packed-verify.cmake)
  tokenomics_packed_verify()
endif()
add_library(tokenomics_core src/policy.cpp src/commit.cpp)
target_include_directories(tokenomics_core PUBLIC src)
target_compile_options(tokenomics_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(tokenomics_runtime src/runtime.cpp)
target_link_libraries(tokenomics_runtime PUBLIC tokenomics_core llama-common)
target_compile_definitions(tokenomics_runtime PUBLIC TOKENOMICS_LLAMA_REV="${LLAMA_REV}")
if(TOKENOMICS_CPU_BATCH_INVARIANT)
  target_compile_definitions(tokenomics_runtime PRIVATE TOKENOMICS_CPU_BATCH_INVARIANT)
endif()
if(TOKENOMICS_CPU_PACKED_VERIFY)
  target_compile_definitions(tokenomics_runtime PRIVATE TOKENOMICS_CPU_PACKED_VERIFY)
endif()
target_compile_options(tokenomics_runtime PRIVATE -Wall -Wextra -Wpedantic)
add_executable(tokenomics src/main.cpp)
target_link_libraries(tokenomics PRIVATE tokenomics_runtime)
target_include_directories(tokenomics PRIVATE third_party/llama.cpp/vendor)
if(TOKENOMICS_BUILD_STOCK_DIAGNOSTIC)
  # Unmodified example, linked to the same guarded backend. Its stopping/count
  # conventions differ from Tokenomics; do not merge its aggregate rate into
  # the paired exact-token endpoint.
  add_subdirectory(third_party/llama.cpp/examples/speculative-simple)
  set_target_properties(llama-speculative-simple PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
endif()
if(BUILD_TESTING)
  add_executable(test_core tests/test_core.cpp)
  target_link_libraries(test_core PRIVATE tokenomics_core)
  add_test(NAME core COMMAND test_core)
  add_executable(test_runtime tests/test_runtime.cpp)
  target_link_libraries(test_runtime PRIVATE tokenomics_runtime)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME fixtures COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/make_fixtures.py --out ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
  set_tests_properties(fixtures PROPERTIES FIXTURES_SETUP models)
  add_test(NAME runtime COMMAND test_runtime ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
  set_tests_properties(runtime PROPERTIES FIXTURES_REQUIRED models)
  add_test(NAME tooling COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_*.py)
  set_tests_properties(tooling PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} FIXTURES_REQUIRED models ENVIRONMENT "TOKENOMICS_TEST_BINARY=$<TARGET_FILE:tokenomics>")
endif()
